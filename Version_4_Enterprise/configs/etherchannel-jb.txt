! EtherChannel (LACP) between Server_Switch and JB_Multilayer_Switch

! ---- Server_Switch ----
enable
configure terminal
interface range fa0/1, fa0/10
 switchport trunk encapsulation dot1q
 switchport trunk native vlan 999
 switchport mode trunk
 switchport nonegotiate
 channel-group 1 mode active
 exit
interface port-channel 1
 description To-JB_Multilayer_Switch-Etherchannel
 switchport trunk encapsulation dot1q
 switchport trunk native vlan 999
 switchport mode trunk
 switchport nonegotiate
 exit
end
write memory

! ---- JB_Multilayer_Switch ----
enable
configure terminal
interface range fa0/2, fa0/4
 switchport trunk encapsulation dot1q
 switchport trunk native vlan 999
 switchport mode trunk
 switchport nonegotiate
 channel-group 1 mode active
 exit
interface port-channel 1
 description To-Server_Switch-Etherchannel
 switchport trunk encapsulation dot1q
 switchport trunk native vlan 999
 switchport mode trunk
 switchport nonegotiate
 exit
end
write memory
